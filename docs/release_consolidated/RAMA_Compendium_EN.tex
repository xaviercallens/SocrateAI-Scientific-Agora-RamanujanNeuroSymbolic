\begin{align*}

  c_{\text{eff}}(0) &= -\sum_d \frac{r_d}{d} \in \mathbb{Q}, \\
  c_{\text{eff}}(c) &= -\frac{1}{c^2} \sum_d \frac{\gcd(d, c)^2}{d} r_d \quad (c \mid N), &
  E_0(\mathbf{r}) &= -\frac{1}{24} \sum_d d r_d \in \mathbb{Q}.
\end{align*}
